import sqlite3
from contextlib import closing

from fastapi.testclient import TestClient

from shortener.app import create_app


def test_rejects_unknown_request_fields(tmp_path):
    client = TestClient(create_app(tmp_path / "links.sqlite"))
    response = client.post("/urls", json={"url": "https://example.com", "unexpected": True})
    assert response.status_code == 422


def test_enables_wal(tmp_path):
    path = tmp_path / "links.sqlite"
    create_app(path)
    with closing(sqlite3.connect(path)) as db:
        assert db.execute("PRAGMA journal_mode").fetchone() == ("wal",)
